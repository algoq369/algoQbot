if (holdTime > MAX_HOLD_TIME) {
  exitReason = 'timeout';

  exitReason = 'stop_loss';

#### Step 2: Update `recordPositionExit()` to accept exitReason

**File:** `agents/TradingStrategyAgent.js`

```javascript

**Fixed Code:**

async recordPositionExit(position, exitPrice, exitReason) {
  const profit = this.calculateProfit(position, exitPrice);
  const holdTime = Date.now() - position.entryTime;
  
  // Record shadow trade with ALL data
  if (global.shadowMode && global.shadowMode.recordPositionExit) {
    await global.shadowMode.recordPositionExit({
      positionId: position.id,
      entryPrice: position.entryPrice,
      exitPrice: exitPrice,
      exitReason: exitReason || 'unknown', // ✅ FIX: Ensure exitReason is set
      profit: profit,
      holdTime: holdTime, // ✅ FIX: Calculate and record holdTime
      strategy: position.strategy || 'unknown', // ✅ FIX: Record strategy
      timestamp: Date.now()
    });
  }
}
```

#### Step 2: Clean existing orphaned exits

**File:** `scripts/cleanup-shadow-trades.js`

**Action:** Add function to remove orphaned exits

**Code:**
```javascript
function removeOrphanedExits(trades) {
  const entries = trades.filter(t => t.type !== 'EXIT');
  const exits = trades.filter(t => t.type === 'EXIT');
  
  // Match exits to entries by positionId or timestamp proximity
  const validExits = exits.filter(exit => {
    const matchingEntry = entries.find(entry => 
      entry.positionId === exit.positionId ||
      (Math.abs(entry.timestamp - exit.timestamp) < 3600000) // Within 1 hour
    );
    return !!matchingEntry;
  });
  
  return [...entries, ...validExits];
}
```

### Issue 3.1: Price History Save Error

**Problem:** `ENOENT: no such file or directory` when renaming

**Root Cause:** Directory might not exist when renaming

**File:** `utils/priceHistoryManager.js`

**Current Code (Line 173):**
```javascript
// Atomic rename
await fs.rename(tempPath, resolvedPath);
```

**Fixed Code:**
```javascript
// ✅ FIX: Ensure directory exists before rename
const dataDir = path.dirname(resolvedPath);
try {
  await fs.mkdir(dataDir, { recursive: true, mode: 0o755 });
} catch (mkdirError) {
  if (mkdirError.code !== 'EEXIST') {
    throw mkdirError; // Re-throw if not "already exists"
  }
}

// Now safe to rename
await fs.rename(tempPath, resolvedPath);
```
